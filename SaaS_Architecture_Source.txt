<?php

namespace App\Http\Controllers;

use App\Models\Product;
use App\Models\Site;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;
use Illuminate\Validation\ValidationException;
use Inertia\Inertia;

class CartController extends Controller
{
    // 取得每個商店獨立的購物車 Session Key
    private function cartKey(Site $site): string
    {
        return 'shop_cart_' . $site->id;
    }

    // 取得商品可購買的規格與庫存
    private function resolveProduct(Site $site, int $productId, ?int $variantId): array
    {
        $product = Product::with('variants')
            ->where('site_id', $site->id)
            ->where('is_active', true)
            ->findOrFail($productId);

        $activeVariants = $product->variants
            ->where('is_active', true);

        if ($activeVariants->isNotEmpty()) {
            if ($variantId === null) {
                throw ValidationException::withMessages([
                    'variant_id' => '請先選擇商品規格。',
                ]);
            }

            $variant = $activeVariants->firstWhere('id', $variantId);

            if (!$variant) {
                throw ValidationException::withMessages([
                    'variant_id' => '商品規格不存在或已停用。',
                ]);
            }

            return [
                'product' => $product,
                'variant' => $variant,
                'price' => (float) $variant->price,
                'stock' => (int) $variant->stock,
            ];
        }

        if ($product->variants->isNotEmpty()) {
            throw ValidationException::withMessages([
                'product_id' => '這項商品目前沒有可購買的規格。',
            ]);
        }

        if ($variantId !== null) {
            throw ValidationException::withMessages([
                'variant_id' => '這項商品不需要選擇規格。',
            ]);
        }

        return [
            'product' => $product,
            'variant' => null,
            'price' => (float) $product->price,
            'stock' => (int) $product->stock,
        ];
    }

    // 建立購物車項目識別碼
    private function itemKey(int $productId, ?int $variantId): string
    {
        return 'p' . $productId . '_v' . ($variantId ?? 0);
    }

    // 顯示購物車並重新檢查價格與庫存
    public function index(Request $request, Site $site)
    {
        $cart = $request->session()->get($this->cartKey($site), []);
        $items = [];
        $total = 0;
        $count = 0;

        foreach ($cart as $key => $row) {
            try {
                $result = $this->resolveProduct(
                    $site,
                    (int) $row['product_id'],
                    isset($row['variant_id']) ? (int) $row['variant_id'] : null
                );
            } catch (\Illuminate\Database\Eloquent\ModelNotFoundException | ValidationException $e) {
                continue;
            }

            $product = $result['product'];
            $variant = $result['variant'];
            $quantity = min((int) $row['quantity'], $result['stock']);

            if ($quantity <= 0) {
                continue;
            }

            $price = $result['price'];
            $subtotal = round($price * $quantity, 2);

            $image = $product->image;
            // 取得商品圖片網址
            $imageUrl = $image
                ? (
                    str_starts_with($image, 'http://') ||
                    str_starts_with($image, 'https://') ||
                    str_starts_with($image, '/')
                        ? $image
                        : asset('storage/' . ltrim($image, '/'))
                )
                : null;
            $items[] = [
                'key' => $key,
                'product_id' => $product->id,
                'variant_id' => $variant?->id,
                'name' => $product->name,
                'options' => $variant?->options ?? [],
                'image_url' => $imageUrl,
                'price' => $price,
                'quantity' => $quantity,
                'stock' => $result['stock'],
                'subtotal' => $subtotal,
            ];

            $total += $subtotal;
            $count += $quantity;
        }

        return Inertia::render('Shop/Cart', [
            'site' => [
                'id' => $site->id,
                'name' => $site->name,
                'slug' => $site->slug,
            ],
            'items' => $items,
            'total' => round($total, 2),
            'count' => $count,
        ]);
    }

    // 加入購物車
    public function store(Request $request, Site $site)
    {
        $data = $request->validate([
            'product_id' => ['required', 'integer'],
            'variant_id' => ['nullable', 'integer'],
            'quantity' => ['required', 'integer', 'min:1'],
        ]);

        $productId = (int) $data['product_id'];
        $variantId = isset($data['variant_id']) ? (int) $data['variant_id'] : null;
        $quantity = (int) $data['quantity'];

        $result = $this->resolveProduct($site, $productId, $variantId);

        $key = $this->itemKey($productId, $variantId);
        $sessionKey = $this->cartKey($site);
        $cart = $request->session()->get($sessionKey, []);

        $existing = (int) ($cart[$key]['quantity'] ?? 0);
        $newQuantity = $existing + $quantity;

        if ($newQuantity > $result['stock']) {
            throw ValidationException::withMessages([
                'quantity' => '加入數量超過可購買庫存。',
            ]);
        }

        $cart[$key] = [
            'product_id' => $productId,
            'variant_id' => $variantId,
            'quantity' => $newQuantity,
        ];

        $request->session()->put($sessionKey, $cart);

        return redirect()->route('shop.cart.index', $site->id);
    }

    // 修改購物車數量
    public function update(Request $request, Site $site, string $key)
    {
        $data = $request->validate([
            'quantity' => ['required', 'integer', 'min:1'],
        ]);

        $sessionKey = $this->cartKey($site);
        $cart = $request->session()->get($sessionKey, []);

        if (!isset($cart[$key])) {
            abort(404);
        }

        $row = $cart[$key];

        $result = $this->resolveProduct(
            $site,
            (int) $row['product_id'],
            isset($row['variant_id']) ? (int) $row['variant_id'] : null
        );

        if ((int) $data['quantity'] > $result['stock']) {
            throw ValidationException::withMessages([
                'quantity' => '購買數量超過可購買庫存。',
            ]);
        }

        $cart[$key]['quantity'] = (int) $data['quantity'];
        $request->session()->put($sessionKey, $cart);

        return back();
    }

    // 移除購物車商品
    public function destroy(Request $request, Site $site, string $key)
    {
        $sessionKey = $this->cartKey($site);
        $cart = $request->session()->get($sessionKey, []);

        unset($cart[$key]);

        $request->session()->put($sessionKey, $cart);

        return back();
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Order;
use App\Models\Product;
use App\Models\ProductVariant;
use App\Models\Site;
use Illuminate\Database\QueryException;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;
use Inertia\Inertia;

class CheckoutController extends Controller
{
    // 取得商店獨立的購物車 Key
    private function cartKey(Site $site): string
    {
        return 'shop_cart_' . $site->id;
    }

    // 取得商店獨立的結帳 Token Key
    private function tokenKey(Site $site): string
    {
        return 'shop_checkout_token_' . $site->id;
    }

    // 取得商店最近成功建立的訂單 Key
    private function lastOrderKey(Site $site): string
    {
        return 'shop_last_order_' . $site->id;
    }

    // 驗證購物車商品、規格、價格及庫存
    private function getCartItems(Site $site, array $cart, bool $lock = false): array
    {
        if (empty($cart)) {
            throw ValidationException::withMessages([
                'cart' => '購物車沒有商品，請先選購。',
            ]);
        }

        $items = [];
        $subtotalCents = 0;

        foreach ($cart as $row) {
            $productId = (int) ($row['product_id'] ?? 0);
            $variantId = isset($row['variant_id'])
                ? (int) $row['variant_id']
                : null;
            $quantity = (int) ($row['quantity'] ?? 0);

            $productQuery = Product::where('site_id', $site->id)
                ->where('is_active', true);

            if ($lock) {
                $productQuery->lockForUpdate();
            }

            $product = $productQuery->find($productId);

            if (!$product || $quantity < 1) {
                throw ValidationException::withMessages([
                    'cart' => '購物車包含無法購買的商品，請返回購物車檢查。',
                ]);
            }

            $hasVariants = $product->variants()->exists();
            $variant = null;
            $price = $product->price;
            $stock = (int) $product->stock;

            if ($hasVariants) {
                if ($variantId === null) {
                    throw ValidationException::withMessages([
                        'cart' => '有商品尚未選擇規格。',
                    ]);
                }

                $variantQuery = ProductVariant::where('product_id', $product->id)
                    ->where('is_active', true);

                if ($lock) {
                    $variantQuery->lockForUpdate();
                }

                $variant = $variantQuery->find($variantId);

                if (!$variant) {
                    throw ValidationException::withMessages([
                        'cart' => '商品規格已停用或不存在。',
                    ]);
                }

                $price = $variant->price;
                $stock = (int) $variant->stock;
            } elseif ($variantId !== null) {
                throw ValidationException::withMessages([
                    'cart' => '購物車商品規格不正確。',
                ]);
            }

            if ($quantity > $stock) {
                throw ValidationException::withMessages([
                    'cart' => "{$product->name} 的庫存不足，請返回購物車調整數量。",
                ]);
            }

            $priceCents = (int) round((float) $price * 100);
            $lineCents = $priceCents * $quantity;
            $subtotalCents += $lineCents;

            $items[] = [
                'product_id' => $product->id,
                'product_variant_id' => $variant?->id,
                'product_name' => $product->name,
                'variant_options' => $variant?->options,
                'unit_price' => number_format($priceCents / 100, 2, '.', ''),
                'quantity' => $quantity,
                'subtotal' => number_format($lineCents / 100, 2, '.', ''),
                'stock_model' => $variant ?? $product,
            ];
        }

        return [
            'items' => $items,
            'subtotal_cents' => $subtotalCents,
        ];
    }

    // 顯示結帳頁並產生本次結帳專用 Token
    public function index(Request $request, Site $site)
    {
        $cart = $request->session()->get($this->cartKey($site), []);

        if (empty($cart)) {
            return redirect()->route('shop.cart.index', $site->id);
        }

        $result = $this->getCartItems($site, $cart);

        $checkoutToken = Str::random(64);

        $request->session()->put(
            $this->tokenKey($site),
            $checkoutToken
        );

        return Inertia::render('Shop/Checkout', [
            'site' => [
                'id' => $site->id,
                'name' => $site->name,
            ],
            'items' => array_map(function ($item) {
                unset($item['stock_model']);

                return $item;
            }, $result['items']),
            'subtotal' => $result['subtotal_cents'] / 100,
            'shippingFee' => 60,
            'total' => ($result['subtotal_cents'] + 6000) / 100,
            'checkoutToken' => $checkoutToken,
        ]);
    }

    // 建立訂單並防止同一 Token 重複下單
    public function store(Request $request, Site $site)
    {
        $data = $request->validate([
            'checkout_token' => ['required', 'string', 'size:64'],
            'customer_name' => ['required', 'string', 'max:100'],
            'customer_phone' => ['required', 'string', 'max:30'],
            'customer_email' => ['required', 'email', 'max:150'],
            'recipient_name' => ['required', 'string', 'max:100'],
            'recipient_phone' => ['required', 'string', 'max:30'],
            'shipping_address' => ['required', 'string', 'max:1000'],
            'customer_note' => ['nullable', 'string', 'max:2000'],
        ]);

        $token = $data['checkout_token'];
        $sessionToken = $request->session()->get($this->tokenKey($site));

        if (!$sessionToken || !hash_equals($sessionToken, $token)) {
            throw ValidationException::withMessages([
                'checkout_token' => '結帳頁面已過期，請重新進入結帳頁。',
            ]);
        }

        // 相同 Token 已建立訂單時直接返回原訂單
        $existingOrder = Order::where('checkout_token', $token)
            ->where('site_id', $site->id)
            ->first();

        if ($existingOrder) {
            $request->session()->put(
                $this->lastOrderKey($site),
                $existingOrder->id
            );

            return redirect()->route('shop.checkout.success', [
                'site' => $site->id,
                'order' => $existingOrder->id,
            ]);
        }

        $cartKey = $this->cartKey($site);
        $cart = $request->session()->get($cartKey, []);

        try {
            // 資料庫交易內重新驗證並鎖定庫存
            $order = DB::transaction(function () use ($site, $cart, $data, $token) {
                $result = $this->getCartItems($site, $cart, true);

                $subtotalCents = $result['subtotal_cents'];
                $shippingCents = 6000;

                $order = Order::create([
                    'site_id' => $site->id,
                    'order_number' => 'ORD-' . now()->format('YmdHis') . '-' . strtoupper(Str::random(8)),
                    'checkout_token' => $token,
                    'customer_name' => $data['customer_name'],
                    'customer_phone' => $data['customer_phone'],
                    'customer_email' => $data['customer_email'],
                    'recipient_name' => $data['recipient_name'],
                    'recipient_phone' => $data['recipient_phone'],
                    'shipping_address' => $data['shipping_address'],
                    'shipping_method' => 'home_delivery',
                    'subtotal' => number_format($subtotalCents / 100, 2, '.', ''),
                    'shipping_fee' => '60.00',
                    'discount_amount' => '0.00',
                    'total' => number_format(
                        ($subtotalCents + $shippingCents) / 100,
                        2,
                        '.',
                        ''
                    ),
                    'payment_method' => 'pending_selection',
                    'payment_status' => 'pending',
                    'order_status' => 'pending',
                    'customer_note' => $data['customer_note'] ?? null,
                    'access_token' => Str::random(64),
                ]);

                foreach ($result['items'] as $item) {
                    $stockModel = $item['stock_model'];
                    unset($item['stock_model']);

                    $order->items()->create($item);

                    $stockModel->stock -= $item['quantity'];
                    $stockModel->save();
                }

                return $order;
            }, 3);
        } catch (QueryException $e) {
            // 同一 Token 的並發請求由資料庫唯一限制擋下
            if ((int) ($e->errorInfo[1] ?? 0) !== 1062) {
                throw $e;
            }

            $order = Order::where('checkout_token', $token)
                ->where('site_id', $site->id)
                ->first();

            if (!$order) {
                throw $e;
            }
        }

        // 交易成功後才清空購物車並記錄訂單
        $request->session()->forget($cartKey);

        $request->session()->put(
            $this->lastOrderKey($site),
            $order->id
        );

        return redirect()->route('shop.checkout.success', [
            'site' => $site->id,
            'order' => $order->id,
        ]);
    }

    // 確認同一商店及目前 Session 才能查看訂單
    public function success(Request $request, Site $site, Order $order)
    {
        if (
            $order->site_id !== $site->id ||
            (int) $request->session()->get($this->lastOrderKey($site)) !== $order->id
        ) {
            abort(404);
        }

        $order->load('items');

        return Inertia::render('Shop/OrderSuccess', [
            'site' => [
                'id' => $site->id,
                'name' => $site->name,
            ],
            'order' => [
                'id' => $order->id,
                'order_number' => $order->order_number,
                'subtotal' => $order->subtotal,
                'shipping_fee' => $order->shipping_fee,
                'total' => $order->total,
                'payment_status' => $order->payment_status,
                'order_status' => $order->order_status,
                'items' => $order->items->map(fn ($item) => [
                    'product_name' => $item->product_name,
                    'variant_options' => $item->variant_options,
                    'unit_price' => $item->unit_price,
                    'quantity' => $item->quantity,
                    'subtotal' => $item->subtotal,
                ]),
            ],
        ]);
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Support\Facades\Gate;
use App\Models\Order;
use App\Models\Product;
use App\Models\ProductVariant;
use App\Models\Site;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\Rule;
use Illuminate\Validation\ValidationException;
use Inertia\Inertia;

class OrderController extends Controller
{
    // 驗證登入者角色與訂單所有權
    private function ownedOrder(int $orderId): Order
    {
        // 驗證使用者是否具有商店管理權限
        Gate::authorize('viewAny', Site::class);

        // 確認訂單屬於目前登入者的商店
        return Order::query()
            ->whereKey($orderId)
            ->whereHas('site', function ($query) {
                $query->where('owner_user_id', Auth::id());
            })
            ->firstOrFail();
    }

    // 顯示商家所屬網站的訂單列表
    public function index(Request $request)
    {
        // 驗證使用者是否具有商店管理權限
        Gate::authorize('viewAny', Site::class);

        $sites = Site::query()
            ->where('owner_user_id', Auth::id())
            ->orderBy('id')
            ->get(['id', 'name', 'slug']);

        $siteIds = $sites->pluck('id');

        $search = trim((string) $request->query('search', ''));
        $selectedSite = $request->query('site_id');

        $query = Order::query()
            ->whereIn('site_id', $siteIds)
            ->with(['site:id,name'])
            ->withCount('items');

        // 依網站篩選訂單
        if ($selectedSite !== null && $selectedSite !== '') {
            $query->where('site_id', $selectedSite);
        }

        // 搜尋訂單編號、訂購人、收件人及電話
        if ($search !== '') {
            $query->where(function ($q) use ($search) {
                $q->where('order_number', 'like', "%{$search}%")
                    ->orWhere('customer_name', 'like', "%{$search}%")
                    ->orWhere('customer_phone', 'like', "%{$search}%")
                    ->orWhere('recipient_name', 'like', "%{$search}%")
                    ->orWhere('recipient_phone', 'like', "%{$search}%");
            });
        }

        $orders = $query
            ->latest()
            ->paginate(15)
            ->withQueryString();

        return Inertia::render('Orders/Index', [
            'sites' => $sites,
            'orders' => $orders,
            'filters' => [
                'search' => $search,
                'site_id' => $selectedSite ?? '',
            ],
        ]);
    }

    // 顯示商家所屬網站的訂單明細
    public function show(Order $order)
    {
        $order = $this->ownedOrder($order->id);

        $order->load(['site:id,name', 'items']);

        return Inertia::render('Orders/Show', [
            'order' => $order,
        ]);
    }

    // 將待處理訂單更新為處理中
    public function process(Order $order)
    {
        $this->ownedOrder($order->id);

        DB::transaction(function () use ($order) {
            $lockedOrder = Order::whereKey($order->id)
                ->lockForUpdate()
                ->firstOrFail();

            if ($lockedOrder->order_status !== 'pending') {
                throw ValidationException::withMessages([
                    'order_status' => '只有待處理訂單可以改為處理中。',
                ]);
            }

            $lockedOrder->update([
                'order_status' => 'processing',
            ]);
        });

        return back()->with('success', '訂單已改為處理中。');
    }


    // 本機模擬付款成功
    public function simulatePayment(Order $order)
    {
        // 正式環境禁止使用
        abort_unless(
            app()->environment(['local', 'testing']),
            404
        );

        // 確認訂單屬於目前登入的商家
        $this->ownedOrder($order->id);

        DB::transaction(function () use ($order) {
            $lockedOrder = Order::whereKey($order->id)
                ->whereHas('site', function ($query) {
                    $query->where('owner_user_id', Auth::id());
                })
                ->lockForUpdate()
                ->firstOrFail();

            // 只允許尚未付款且尚未出貨的訂單
            if (
                !in_array(
                    $lockedOrder->order_status,
                    ['pending', 'processing'],
                    true
                ) ||
                $lockedOrder->payment_status !== 'pending'
            ) {
                throw ValidationException::withMessages([
                    'payment_status' => '只有尚未付款且尚未出貨的訂單可以模擬付款。',
                ]);
            }

            $lockedOrder->update([
                'payment_status' => 'paid',
            ]);
        }, 3);

        return back()->with('success', '測試付款成功，訂單已標記為已付款。');
    }


    // 登記訂單出貨資訊
    public function ship(Request $request, Order $order)
    {
        $this->ownedOrder($order->id);

        $data = $request->validate([
            'shipping_company' => ['required', 'string', 'max:100'],
            'tracking_number' => ['required', 'string', 'max:100'],
        ]);

        DB::transaction(function () use ($order, $data) {
            $lockedOrder = Order::whereKey($order->id)
                ->lockForUpdate()
                ->firstOrFail();

            if ($lockedOrder->order_status !== 'processing') {
                throw ValidationException::withMessages([
                    'order_status' => '只有處理中的訂單可以登記出貨。',
                ]);
            }

            if ($lockedOrder->payment_status !== 'paid') {
                throw ValidationException::withMessages([
                    'payment_status' => '此訂單尚未付款，不能登記出貨。',
                ]);
            }

            $lockedOrder->update([
                'order_status' => 'shipped',
                'shipping_company' => $data['shipping_company'],
                'tracking_number' => $data['tracking_number'],
                'shipped_at' => now(),
            ]);
        });

        return back()->with('success', '訂單已登記出貨。');
    }

    // 將已出貨訂單標記為完成
    public function complete(Order $order)
    {
        $this->ownedOrder($order->id);

        DB::transaction(function () use ($order) {
            $lockedOrder = Order::whereKey($order->id)
                ->lockForUpdate()
                ->firstOrFail();

            if (
                $lockedOrder->order_status !== 'shipped' ||
                $lockedOrder->payment_status !== 'paid'
            ) {
                throw ValidationException::withMessages([
                    'order_status' => '只有已付款且已出貨的訂單可以完成。',
                ]);
            }

            $lockedOrder->update([
                'order_status' => 'completed',
            ]);
        });

        return back()->with('success', '訂單已完成。');
    }

    // 取消未付款訂單並恢復商品庫存
    public function cancel(Order $order)
    {
        $this->ownedOrder($order->id);

        DB::transaction(function () use ($order) {
            $lockedOrder = Order::whereKey($order->id)
                ->lockForUpdate()
                ->firstOrFail();

            if (!in_array(
                $lockedOrder->order_status,
                ['pending', 'processing'],
                true
            )) {
                throw ValidationException::withMessages([
                    'order_status' => '只有尚未出貨且未取消的訂單可以取消。',
                ]);
            }

            if ($lockedOrder->payment_status !== 'pending') {
                throw ValidationException::withMessages([
                    'payment_status' => '已付款或付款異常的訂單需進入退款流程，不能直接取消。',
                ]);
            }

            $items = $lockedOrder->items()
                ->orderBy('product_id')
                ->orderBy('product_variant_id')
                ->get();

            if ($items->isEmpty()) {
                throw ValidationException::withMessages([
                    'order_status' => '訂單沒有商品明細，無法安全恢復庫存。',
                ]);
            }

            // 逐筆鎖定商品並恢復對應庫存
            foreach ($items as $item) {
                $product = Product::whereKey($item->product_id)
                    ->lockForUpdate()
                    ->first();

                if (!$product || $product->site_id !== $lockedOrder->site_id) {
                    throw ValidationException::withMessages([
                        'order_status' => '訂單商品不存在或網站歸屬不正確，需人工處理庫存。',
                    ]);
                }

                if ($item->product_variant_id !== null) {
                    $variant = ProductVariant::query()
                        ->where('product_id', $product->id)
                        ->whereKey($item->product_variant_id)
                        ->lockForUpdate()
                        ->first();

                    if (!$variant) {
                        throw ValidationException::withMessages([
                            'order_status' => '訂單規格已不存在，需人工處理庫存。',
                        ]);
                    }

                    $variant->increment('stock', $item->quantity);
                } else {
                    $product->increment('stock', $item->quantity);
                }
            }

            // 只有庫存恢復成功才將訂單標記為取消
            $lockedOrder->update([
                'order_status' => 'cancelled',
            ]);
        }, 3);

        return back()->with('success', '訂單已取消，庫存已恢復。');
    }

class ProductVariantController extends Controller
{
    // 驗證使用者角色與商品所有權
    private function checkOwnership(Product $product): void
    {
        // 驗證使用者是否具有商店管理權限
        Gate::authorize('viewAny', Site::class);

        // 驗證商品是否屬於目前登入者的商店
        Site::where('owner_user_id', Auth::id())
            ->findOrFail($product->site_id);
    }


    // 驗證規格組合
    private function validateOptions(array $options): array
    {
        $cleaned = [];

        foreach ($options as $key => $value) {
            if (
                !is_string($key) ||
                trim($key) === '' ||
                mb_strlen($key) > 50 ||
                !is_string($value) ||
                trim($value) === '' ||
                mb_strlen($value) > 100
            ) {
                throw ValidationException::withMessages([
                    'options' => '規格名稱或內容格式不正確',
                ]);
            }

            $cleaned[trim($key)] = trim($value);
        }

        if (count($cleaned) !== count($options)) {
            throw ValidationException::withMessages([
                'options' => '規格名稱不可重複',
            ]);
        }

        ksort($cleaned);

        return $cleaned;
    }

    // 檢查相同商品是否已有相同規格
    private function checkDuplicate(
        Product $product,
        array $options,
        ?int $exceptId = null
    ): void {
        $query = $product->variants();

        if ($exceptId !== null) {
            $query->whereKeyNot($exceptId);
        }

        foreach ($query->get() as $variant) {
            $existing = $variant->options ?? [];
            ksort($existing);

            if ($existing === $options) {
                throw ValidationException::withMessages([
                    'options' => '此商品已存在相同的規格組合',
                ]);
            }
        }
    }

    // 新增商品規格
    public function store(Request $request, Product $product)
    {
        $this->checkOwnership($product);

        $validated = $request->validate([
            'sku' => [
                'nullable',
                'string',
                'max:100',
                Rule::unique('product_variants', 'sku')
                    ->where('product_id', $product->id),
            ],
            'options' => 'required|array|min:1',
            'price' => 'required|numeric|min:0|max:99999999.99',
            'stock' => 'required|integer|min:0|max:4294967295',
            'is_active' => 'required|boolean',
        ]);

        $options = $this->validateOptions(
            $validated['options']
        );

        $this->checkDuplicate($product, $options);

        $product->variants()->create([
            'sku' => $validated['sku'] ?: null,
            'options' => $options,
            'price' => $validated['price'],
            'stock' => $validated['stock'],
            'is_active' => $validated['is_active'],
        ]);

        return redirect()->route('products.index')
            ->with('success', '商品規格新增成功');
    }

    // 修改商品規格
    public function update(
        Request $request,
        Product $product,
        ProductVariant $variant
    ) {
        $this->checkOwnership($product);

        abort_unless(
            $variant->product_id === $product->id,
            404
        );

        $validated = $request->validate([
            'sku' => [
                'nullable',
                'string',
                'max:100',
                Rule::unique('product_variants', 'sku')
                    ->where('product_id', $product->id)
                    ->ignore($variant->id),
            ],
            'options' => 'required|array|min:1',
            'price' => 'required|numeric|min:0|max:99999999.99',
            'stock' => 'required|integer|min:0|max:4294967295',
            'is_active' => 'required|boolean',
        ]);

        $options = $this->validateOptions(
            $validated['options']
        );

        $this->checkDuplicate(
            $product,
            $options,
            $variant->id
        );

        $variant->update([
            'sku' => $validated['sku'] ?: null,
            'options' => $options,
            'price' => $validated['price'],
            'stock' => $validated['stock'],
            'is_active' => $validated['is_active'],
        ]);

        return redirect()->route('products.index')
            ->with('success', '商品規格修改成功');
    }

    // 刪除商品規格
    public function destroy(
        Product $product,
        ProductVariant $variant
    ) {
        $this->checkOwnership($product);

        abort_unless(
            $variant->product_id === $product->id,
            404
        );

        $variant->delete();

        return redirect()->route('products.index')
            ->with('success', '商品規格刪除成功');
    }

const isLocalEnvironment = import.meta.env.DEV;
function simulatePayment() {
    if (!isLocalEnvironment || processing.value) return;
    if (!window.confirm('確定要模擬付款成功嗎？此操作僅供本機測試。')) {
        return;
    }
    processing.value = true;
    router.patch(`${orderUrl.value}/simulate-payment`, {}, {
        preserveScroll: true,
        onFinish: () => {
            processing.value = false;
        },
    });
}



const orderErrors = computed(
    () => page.props.errors as Record<string, string>
);

// 更新訂單處理狀態
function processOrder() {
    if (processing.value) return;

    processing.value = true;

    router.patch(`${orderUrl.value}/process`, {}, {
        preserveScroll: true,
        onFinish: () => processing.value = false,
    });
}

// 登記出貨
function shipOrder() {
    if (processing.value) return;

    processing.value = true;

    router.patch(`${orderUrl.value}/ship`, {
        shipping_company: shippingCompany.value,
        tracking_number: trackingNumber.value,
    }, {
        preserveScroll: true,
        onFinish: () => processing.value = false,
    });
}

// 完成訂單
function completeOrder() {
    if (processing.value) return;

    processing.value = true;

    router.patch(`${orderUrl.value}/complete`, {}, {
        preserveScroll: true,
        onFinish: () => processing.value = false,
    });
}

// 取消訂單並恢復庫存
function cancelOrder() {
    if (processing.value) return;

    if (!window.confirm('確定取消這筆訂單並恢復庫存嗎？')) {
        return;
    }

    processing.value = true;

    router.patch(`${orderUrl.value}/cancel`, {}, {
        preserveScroll: true,
        onFinish: () => processing.value = false,
    });
}

const imageError = ref('');
const editImageError = ref('');

const allowedTypes = ['image/jpeg', 'image/png', 'image/webp'];
const maxImageSize = 5 * 1024 * 1024;

function imageUrl(path: string | null): string | null {
    return path ? '/storage/' + path : null;
}

function clearPreview(url: string | null) {
    if (url?.startsWith('blob:')) {
        URL.revokeObjectURL(url);
    }
}

function validateImage(file: File): string {
    if (!allowedTypes.includes(file.type)) {
        return '只能上傳 JPG、PNG 或 WebP 圖片';
    }

    if (file.size > maxImageSize) {
        return '圖片不能超過 5MB';
    }

    return '';
}

// 新增商品
const form = useForm({
    site_id: props.sites[0]?.id ?? '',
    name: '',
    description: '',
    price: '' as string | number,
    stock: 0,
    image: null as File | null,
});

function onImageChange(event: Event) {
    const input = event.target as HTMLInputElement;
    const file = input.files?.[0];

    if (!file) return;

    imageError.value = validateImage(file);

    if (imageError.value) {
        input.value = '';
        return;
    }

    clearPreview(newImagePreview.value);

    form.image = file;
    newImagePreview.value = URL.createObjectURL(file);
    form.clearErrors('image');
}

function submit() {
    if (imageError.value) return;

    form.post('/products', {
        forceFormData: true,
        onSuccess: () => {
            form.reset('name', 'description', 'price', 'stock', 'image');

            clearPreview(newImagePreview.value);
            newImagePreview.value = null;

            imageError.value = '';

            if (imageInput.value) {
                imageInput.value.value = '';
            }
        },
    });
}

function startEdit(product: Product) {

const managingProductId = ref<number | null>(null);

function resetVariantForm() {
    editingVariantId.value = null;
    variantMessage.value = '';

    variantForm.reset();
    variantForm.clearErrors();

    optionRows.value = [
        { name: '顏色', value: '' },
    ];

function toggleVariantManager(product: Product) {
    // 再次點擊時關閉規格管理
    if (managingProductId.value === product.id) {
        managingProductId.value = null;
        resetVariantForm();
        return;
    }

    // 開啟規格管理前先關閉商品編輯
    cancelEdit();

    managingProductId.value = product.id;
    resetVariantForm();
}


function addOptionRow() {
    optionRows.value.push({
        name: '',
        value: '',
    });
}

function removeOptionRow(index: number) {
    if (optionRows.value.length <= 1) return;

    optionRows.value.splice(index, 1);
}

function startVariantEdit(variant: ProductVariant) {
    editingVariantId.value = variant.id;
    variantMessage.value = '';

    variantForm.sku = variant.sku ?? '';
    variantForm.price = variant.price;
    variantForm.stock = variant.stock;
    variantForm.is_active = Boolean(variant.is_active);

    optionRows.value = Object.entries(variant.options).map(
        ([name, value]) => ({
            name,
            value,
        })
    );

    variantForm.clearErrors();
}

function buildOptions(): Record<string, string> | null {
    const options: Record<string, string> = Object.create(null);

    for (const row of optionRows.value) {
        const name = row.name.trim();
        const value = row.value.trim();

        if (!name || !value) {
            variantMessage.value = '規格名稱與內容不可空白';
            return null;
        }

        if (Object.prototype.hasOwnProperty.call(options, name)) {
            variantMessage.value = '規格名稱不可重複';
            return null;
        }

        options[name] = value;
    }

    return options;
}

// 新增或修改商品規格
function saveVariant(productId: number) {
    variantMessage.value = '';

    const options = buildOptions();

    if (!options) return;

    variantForm.options = options;

    if (editingVariantId.value !== null) {
        variantForm.put(
            `/products/${productId}/variants/${editingVariantId.value}`,
            {
                preserveScroll: true,
                onSuccess: () => resetVariantForm(),
            }
        );

        return;
    }

    variantForm.post(`/products/${productId}/variants`, {
        preserveScroll: true,
        onSuccess: () => resetVariantForm(),
    });
}

// 刪除商品規格
function deleteVariant(productId: number, variant: ProductVariant) {
    if (!confirm('確定刪除此商品規格嗎？')) {
        return;
    }

    router.delete(
        `/products/${productId}/variants/${variant.id}`,
        {
            preserveScroll: true,
            onSuccess: () => {
                if (editingVariantId.value === variant.id) {
                    resetVariantForm();
                }
            },
        }
    );
}

// 規格顯示文字
function variantLabel(options: Record<string, string>): string {
    return Object.entries(options)
        .map(([name, value]) => `${name}：${value}`)
        .join(' / ');
}

// 規格總庫存
function totalVariantStock(product: Product): number {
    return (product.variants ?? [])
        .filter(variant => Boolean(variant.is_active))
        .reduce((total, variant) => total + Number(variant.stock), 0);
}

type CartItem = {
    key: string
    product_id: number
    variant_id: number | null

    slug: string

    site: Shop

    total: number
    count: number

const cartUrl = computed(() => `${baseUrl.value}/cart`)

const money = (value: number) =>
    new Intl.NumberFormat('zh-TW', {
        style: 'currency',
        currency: 'TWD',
        maximumFractionDigits: 0,
    }).format(value)

const optionsText = (options: Record<string, string>) =>
    Object.entries(options)
        .map(([key, value]) => `${key}：${value}`)
        .join(' / ')

// 修改購買數量
function updateQuantity(item: CartItem, quantity: number) {
    if (processing.value || quantity < 1 || quantity > item.stock) {
        return
    }

    processing.value = true

    router.put(
        `${cartUrl.value}/${encodeURIComponent(item.key)}`,
        { quantity },
        {
            preserveScroll: true,
            onFinish: () => {
                processing.value = false
            },
        },
    )
}

// 刪除購物車商品
function removeItem(item: CartItem) {
    if (processing.value) return

    processing.value = true

    router.delete(
        `${cartUrl.value}/${encodeURIComponent(item.key)}`,
        {
            preserveScroll: true,
            onFinish: () => {
                processing.value = false
            },
        },
    )
}

const money = (value: string | number) => {
    return `NT$ ${Number(value).toLocaleString('zh-TW')}`;
};

// 將訂購人資訊同步至收件人
function syncRecipient() {
    if (sameRecipient.value) {
        form.recipient_name = form.customer_name;
        form.recipient_phone = form.customer_phone;
    }
}

// 顯示商品規格

    if (!options) return '';

    return Object.entries(options)
        .map(([key, value]) => `${key}：${value}`)
        .join(' / ');
}

    syncRecipient();

    });
}

function displayPrice(product: Product): string {
    if (!product.variants?.length) {
        return money(product.price);
    }

    const prices = product.variants.map(v => Number(v.price));
    const min = Math.min(...prices);
    const max = Math.max(...prices);

    return min === max
        ? money(min)
        : `${money(min)} ～ ${money(max)}`;
}

// 商品庫存計算
function availableStock(product: Product): number {
    if (!product.variants?.length) {
        return Number(product.stock);
    }

    return product.variants.reduce(
        (sum, variant) => sum + Number(variant.stock),
        0
    );
}

function money(value: string | number): string {

    options: Record<string, string>;

    stock: number;
    is_active: boolean | number;
}

interface Product {
    id: number;
    name: string;
    description: string | null;
    image: string | null;
    price: string | number;
    stock: number;
    variants: Variant[];
}

interface Site {
    id: number;
    name: string;
}

// 接收商品詳細資料
const props = defineProps<{
    site: Site;
    product: Product;
}>();

const page = usePage();

// 商品規格及數量狀態
const availableVariants = computed(() => {
    return (props.product.variants ?? []).filter(
        variant => Boolean(variant.is_active)
    );
});

const firstAvailableVariant = availableVariants.value.find(
    variant => Number(variant.stock) > 0
);

const selectedVariantId = ref<number | null>(
    firstAvailableVariant?.id ?? null
);

const quantity = ref(1);
const processing = ref(false);

const errorMessage = ref('');

// 商品規格判斷
const hasVariants = computed(() => availableVariants.value.length > 0);

const selectedVariant = computed(() => {
    return availableVariants.value.find(
        variant => variant.id === selectedVariantId.value
    ) ?? null;
});

// 商品價格及庫存
const currentPrice = computed(() => {
    return selectedVariant.value?.price ?? props.product.price;
});

const currentStock = computed(() => {
    if (hasVariants.value) {
        return Number(selectedVariant.value?.stock ?? 0);
    }

    return Number(props.product.stock);
});

const formattedPrice = computed(() => {
    return `NT$ ${Number(currentPrice.value).toLocaleString('zh-TW')}`;
});

const formattedSubtotal = computed(() => {
    const subtotal = Number(currentPrice.value) * quantity.value;
    return `NT$ ${subtotal.toLocaleString('zh-TW')}`;
});

// 商品圖片網址
const imageUrl = computed(() => {
    const image = props.product.image;

    if (!image) return null;

    if (
        image.startsWith('http://') ||
        image.startsWith('https://') ||
        image.startsWith('/')
    ) {
        return image;
    }

    return `/storage/${image}`;
});

const shopUrl = `/shop/${props.site.id}`;
const cartUrl = `${shopUrl}/cart`;

// 商品規格文字
function variantLabel(variant: Variant): string {
    return Object.entries(variant.options ?? {})
        .map(([name, value]) => `${name}：${value}`)
        .join(' / ');
}

// 切換規格後重設購買數量
watch(selectedVariantId, () => {
    quantity.value = 1;
    errorMessage.value = '';
});

// 增加購買數量
function increaseQuantity() {
    if (processing.value) return;

    if (quantity.value < currentStock.value) {
        quantity.value++;
    }
}

// 減少購買數量
function decreaseQuantity() {
    if (processing.value) return;

    if (quantity.value > 1) {
        quantity.value--;
    }
}

// 加入購物車
function addToCart() {
    errorMessage.value = '';

    if (processing.value) return;

    if (currentStock.value <= 0) {
        errorMessage.value = '這項商品目前沒有庫存。';
        return;
    }

    if (hasVariants.value && selectedVariantId.value === null) {
        errorMessage.value = '請先選擇商品規格。';
        return;
    }

    if (
        !Number.isInteger(quantity.value) ||
        quantity.value < 1 ||
        quantity.value > currentStock.value
    ) {
        errorMessage.value = '購買數量不正確或超過可購買庫存。';
        return;
    }

    processing.value = true;

    router.post(
        cartUrl,
        {
            product_id: props.product.id,
            variant_id: hasVariants.value
                ? selectedVariantId.value
                : null,
            quantity: quantity.value,
        },
        {
            preserveScroll: true,

            onError: (errors) => {
                errorMessage.value =
                    errors.quantity ||
                    errors.variant_id ||
                    errors.product_id ||
                    '加入購物車失敗，請稍後再試。';
            },

            onFinish: () => {
                processing.value = false;
            },
        }
    );
}

// 是否可以加入購物車
const canAddToCart = computed(() => {
    return (
        currentStock.value > 0 &&
        (!hasVariants.value || selectedVariantId.value !== null) &&
        !processing.value
    );

<script setup lang="ts">
import { Head, Link, router } from '@inertiajs/vue3';
import { ref } from 'vue';

interface Site {
    id: number;
    name: string;
    slug: string;
    site_type: string;
    status: string;
    domain: string | null;
}

// 接收網站資料
defineProps<{
    sites: Site[];
}>();

// 正在發布的商店 ID
const publishingSiteId = ref<number | null>(null);

// 發布商店
function publishSite(site: Site) {
    if (site.status !== 'draft') {
        return;
    }

    const confirmed = window.confirm(
        `確定要發布「${site.name}」嗎？發布後買家就能瀏覽並購買商品。`
    );

    if (!confirmed) {
        return;
    }

    publishingSiteId.value = site.id;

    router.patch(
        `/sites/${site.id}/publish`,
        {},
        {
            preserveScroll: true,

            onError: (errors) => {
                alert(
                    Object.values(errors).join('\n') ||
                    '發布失敗，請稍後再試。'
                );
            },

            onFinish: () => {
                publishingSiteId.value = null;
            },
        }
    );
}
</script>

        ->name('shop.cart.index');

    // 加入購物車
    Route::post('/shop/{site}/cart', [CartController::class, 'store'])
        ->name('shop.cart.store');

    // 修改購物車商品數量
    Route::put('/shop/{site}/cart/{key}', [CartController::class, 'update'])
        ->name('shop.cart.update');

    // 刪除購物車商品
    Route::delete('/shop/{site}/cart/{key}', [CartController::class, 'destroy'])
        ->name('shop.cart.destroy');

    // 顯示結帳頁面
    Route::get('/shop/{site}/checkout', [CheckoutController::class, 'index'])
        ->name('shop.checkout.index');

    // 送出訂單
    Route::post('/shop/{site}/checkout', [CheckoutController::class, 'store'])
        ->name('shop.checkout.store');

    // 顯示訂單成功頁
    Route::get('/shop/{site}/orders/{order}/success', [
        CheckoutController::class,
        'success'
    ])->name('shop.checkout.success');

});
